1.Determinar se um número é par ou ímpar e positivo ou negativo 

#include <stdio.h>

int main(void)
{
    int num;
    printf ("digite um número: ");
    scanf ("%i", &num);
    if (num % 2 == 0){
        printf("\n%i, é um numero par\n", num);
    }
    else {
        printf("%i, é um numero impar\n", num);
    }
    if (num < 0){
        printf("%i, é um numero negativo", num);
    }
    else {
        printf("%i, é um numero positivo", num);
    }
    return 0;
}

{

    scanf("%f", &n2);
    printf("digite a terceira nota: ");
    scanf("%f", &n3);
    
    media = (n1 + n2 + n3) / 3;
    if (media == 10){
        printf ("Aluno aprovado com louvor! Parabens! nota: %.2f", media);
    }
    else if (media >= 7){
        printf ("Aluno aprovado! Parabéns! nota: %.2f", media);

        printf("Aluno reprovado! Temos que conversar! nota: %.2f", media);

3.Fazer um algoritmo que: • Leia um número indeterminado de linhas contendo cada uma a idade de um indivíduo. • A úl ma linha que não entrará nos cálculos, contém o valor da idade igual a zero. • Calcule e escreva a idade média deste grupo de indivíduos. • Escreva também a maior idade e a menor 

#include <stdio.h>

int main()
{
    float media;
    int idade, menor, maior, soma, cont;
    
    soma = 0;
    menor = 99999;
    maior = 0;
    media = 0;
    cont = 0;
    while(idade != 0){
        printf("Escreva uma idade: ");
        scanf ("%i", &idade);
       
        soma = soma + idade;    
        if(idade < menor && idade != 0){
            menor = idade;
        }
        else if(idade > maior){
            maior = idade;
        }
        if(idade != 0){
            cont++;
        }
    }
    media = soma / cont;
    printf("A media das idades é: %.2f\n", media);
    printf("A menor idade é: %i\n", menor);
    printf("A maior idade é: %i\n", maior);
    return 0;
}

8. Número primo é aquele que só é divisível por ele mesmo e pela unidade. Fazer um algoritmo que determine e escreva os números primos compreendidos entre um intervalo fornecido pelo usuário. 

#include <stdio.h>

int main(void)
{
    int a, b, aux, qtd_divisores;
    a = 0;
    b = 0;
    aux = 0;
    qtd_divisores = 0;
    
    printf("Informe o valor inicial: ");
    scanf("%i", &a);
    printf("Informe o valor final: ");
    scanf("%i", &b);
    
    if(b < a){
        printf("O valor final nao pode ser menor que o valor inicial");
        return 0;
    }
    while( a < b){
        aux = 1;
        qtd_divisores = 0;
        while (aux <= a){
            if ((a % aux) == 0){
                qtd_divisores++;
            }
            aux++;
        }
        if(qtd_divisores == 2){
            printf("o numero %i é um numero primo\n", a);
        }
        a++;
    }
    return 0;
}

9. Anacleto tem 1,50 m e cresce 2 cenmetros por ano, enquanto Felisberto tem 1,10 m e cresce 3 cen metros por ano. Construa um programa que calcule e apresente quantos anos serão necessários para que Felisberto seja mais alto que Anacleto. 

#include <stdio.h>

int main(void)
{
   int anacleto, felisberto, anos;
   
   anacleto = 150;
   felisberto = 110;
   anos = 0;
   
   while (felisberto <= anacleto){
       anacleto = anacleto + 2;
       felisberto = felisberto + 3;
       anos++;
   }
    printf ("Sao necessarios %i anos", anos);
    return 0;
}

10. Fazer um algoritmo para calcular e escrever a seguinte soma:  37x38/1 + 36x37/2 + 35x36/2 + ... + 1x2/37

#include <stdio.h>

int main(void)
{
   int i;
   float soma;
   
   soma = 0; 
   i = 1;
   while (i <= 37){
       soma = soma + ((37 - i + 1) * (38 - i + 1)) / i;
       i++;
   }
    printf("soma é: %.2f", soma);
}
